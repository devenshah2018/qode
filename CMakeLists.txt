cmake_minimum_required(VERSION 3.20)
project(qode VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(QODE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(QODE_SANITIZERS "address,undefined" CACHE STRING "Sanitizers used by QODE_SANITIZE")
option(QODE_FUZZ "Build the Clang libFuzzer parser target" OFF)
add_library(qode_core src/scanner.c src/compiler.c src/runtime.c src/json.c src/sample.c)
set_target_properties(qode_core PROPERTIES OUTPUT_NAME qode)
target_include_directories(qode_core PUBLIC include PRIVATE src)
if(NOT MSVC)
  target_compile_options(qode_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(qode_core PUBLIC m)
endif()
if(QODE_SANITIZE)
  add_compile_options(-fsanitize=${QODE_SANITIZERS} -fno-omit-frame-pointer)
  target_compile_options(qode_core PRIVATE -fsanitize=${QODE_SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${QODE_SANITIZERS})
endif()
if(EMSCRIPTEN)
  add_executable(qode_wasm src/wasm.c)
  target_link_libraries(qode_wasm PRIVATE qode_core)
  set_target_properties(qode_wasm PROPERTIES OUTPUT_NAME qode SUFFIX .js)
  target_link_options(qode_wasm PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1
    -sEXPORT_NAME=createQode -sENVIRONMENT=worker,node -sALLOW_MEMORY_GROWTH=1
    -sMAXIMUM_MEMORY=134217728 -sSTACK_SIZE=1048576 -sFILESYSTEM=0 -sABORTING_MALLOC=0
    "-sEXPORTED_FUNCTIONS=['_qode_web_compile','_qode_web_step','_qode_web_reset','_qode_web_state','_qode_web_program','_qode_web_sample','_qode_web_clone','_qode_web_restore','_qode_web_bytes','_qode_web_drop','_qode_web_destroy','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['ccall','UTF8ToString','getValue','stringToUTF8']")
else()
  add_executable(qode src/cli.c)
  target_link_libraries(qode PRIVATE qode_core)
  if(NOT MSVC)
    target_compile_options(qode PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_executable(qode_bench tools/benchmark.c)
  target_link_libraries(qode_bench PRIVATE qode_core)
  add_executable(qode_embed examples/embed.c)
  target_link_libraries(qode_embed PRIVATE qode_core)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(qode_tests tests/test_core.c)
    target_link_libraries(qode_tests PRIVATE qode_core)
    add_test(NAME core COMMAND qode_tests)
    add_test(NAME embed COMMAND qode_embed)
    add_executable(qode_fuzz_smoke tests/fuzz_smoke.c)
    target_link_libraries(qode_fuzz_smoke PRIVATE qode_core)
    add_test(NAME parser_mutations COMMAND qode_fuzz_smoke)
    set_tests_properties(parser_mutations PROPERTIES TIMEOUT 60)
    if(UNIX)
    add_library(qode_failure_core OBJECT src/scanner.c src/compiler.c src/runtime.c src/json.c src/sample.c)
    target_include_directories(qode_failure_core PRIVATE include src)
    target_compile_definitions(qode_failure_core PRIVATE malloc=qode_test_malloc calloc=qode_test_calloc realloc=qode_test_realloc)
    add_executable(qode_allocation_tests tests/test_allocations.c $<TARGET_OBJECTS:qode_failure_core>)
    target_include_directories(qode_allocation_tests PRIVATE include)
    if(NOT MSVC)
      target_link_libraries(qode_allocation_tests PRIVATE m)
    endif()
    add_test(NAME allocations COMMAND qode_allocation_tests)
    set_tests_properties(allocations PROPERTIES TIMEOUT 60)
    endif()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:qode>)
    add_test(NAME documentation_examples COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_doc_examples.py $<TARGET_FILE:qode>)
    set_tests_properties(core cli documentation_examples PROPERTIES TIMEOUT 60)
    if(UNIX)
      add_test(NAME make_recipes COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_make.py)
      set_tests_properties(make_recipes PROPERTIES TIMEOUT 60)
    endif()
  endif()
  install(TARGETS qode qode_core)
  install(FILES include/qode.h DESTINATION include)
endif()
if(QODE_FUZZ)
  target_compile_options(qode_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(qode_core INTERFACE -fsanitize=address,undefined)
  add_executable(qode_fuzz tests/fuzz_compile.c)
  target_compile_options(qode_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(qode_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(qode_fuzz PRIVATE qode_core)
endif()
